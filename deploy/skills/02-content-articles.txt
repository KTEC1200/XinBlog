# 技能包 02：内容 - 文章

skill = article.list
name = 文章列表
visible = true
handler = listAdminPosts
desc = 分页查询站点文章（含状态、标题、摘要、浏览量、阅读时长），可用 keyword 按标题/摘要搜索。参数 page、limit、keyword。
params = {"type":"object","properties":{"page":{"type":"integer","minimum":1,"description":"页码，默认 1"},"limit":{"type":"integer","minimum":1,"maximum":100,"description":"每页数量，默认 10"},"keyword":{"type":"string","description":"按标题或摘要关键词搜索"}},"additionalProperties":false}

skill = article.read
name = 文章详情
visible = true
handler = getAdminPost
desc = 读取单篇文章的完整内容（正文、摘要、封面、状态、浏览量、作者）。参数 id 必填。
params = {"type":"object","properties":{"id":{"type":"integer","description":"文章 ID"}},"required":["id"],"additionalProperties":false}
